cmake_minimum_required(VERSION 3.21)
project(static_search_tree LANGUAGES CXX)

add_library(static_search_tree INTERFACE)
add_library(sst::static_search_tree ALIAS static_search_tree)
target_include_directories(static_search_tree INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(static_search_tree INTERFACE cxx_std_20)

if(PROJECT_IS_TOP_LEVEL)
  include(CTest)
endif()
option(SST_BUILD_EXAMPLES "Build the usage example" ${PROJECT_IS_TOP_LEVEL})
option(SST_TEST_AVX2 "Test accelerated backends when the host supports them" OFF)
option(SST_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

function(sst_configure_target target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /Zc:__cplusplus)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
  endif()
  if(SST_ENABLE_SANITIZERS)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      message(FATAL_ERROR "Sanitizer configuration requires GCC or Clang")
    endif()
    target_compile_options(${target} PRIVATE -O1 -g -fsanitize=address,undefined
      -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    target_compile_definitions(${target} PRIVATE _GLIBCXX_ASSERTIONS)
  endif()
endfunction()

if(SST_BUILD_EXAMPLES)
  add_executable(sst_example examples/simple_search.cpp)
  target_link_libraries(sst_example PRIVATE sst::static_search_tree)
  sst_configure_target(sst_example)
endif()

if(BUILD_TESTING AND PROJECT_IS_TOP_LEVEL)
  find_package(Threads REQUIRED)
  function(sst_add_test name source)
    add_executable(sst_${name} ${source} ${ARGN})
    target_link_libraries(sst_${name} PRIVATE sst::static_search_tree Threads::Threads)
    sst_configure_target(sst_${name})
    add_test(NAME ${name} COMMAND sst_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 180)
  endfunction()
  foreach(name IN ITEMS reference eytzinger stree robustness workload_data batch api)
    sst_add_test(${name} tests/${name}_test.cpp)
  endforeach()
  sst_add_test(header_integration tests/header_integration_test.cpp tests/header_integration_other.cpp)
  if(TARGET sst_example)
    add_test(NAME example COMMAND sst_example)
  endif()

  # This target can see only the copied header, not the source include directory.
  configure_file(include/stree.hpp copied_header/stree.hpp COPYONLY)
  add_executable(sst_copied_header examples/simple_search.cpp)
  target_include_directories(sst_copied_header PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/copied_header")
  sst_configure_target(sst_copied_header)
  add_test(NAME copied_header COMMAND sst_copied_header)

  if(SST_TEST_AVX2)
    if(NOT CMAKE_CROSSCOMPILING AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang"
       AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64|i.86)$")
      include(CheckCXXSourceRuns)
      check_cxx_source_runs("int main() { __builtin_cpu_init();
        return __builtin_cpu_supports(\"avx2\") && __builtin_cpu_supports(\"popcnt\") ? 0 : 1; }"
        SST_HOST_HAS_AVX2_POPCNT)
      if(SST_HOST_HAS_AVX2_POPCNT)
        foreach(backend IN ITEMS avx2 mask_disabled software_popcount prefetch_disabled)
          foreach(name IN ITEMS stree batch api)
            sst_add_test(${name}_${backend} tests/${name}_test.cpp)
            target_compile_options(sst_${name}_${backend} PRIVATE -mavx2 -mpopcnt)
            if(backend STREQUAL "mask_disabled")
              target_compile_definitions(sst_${name}_${backend} PRIVATE SST_DISABLE_MASK_SIMD)
            elseif(backend STREQUAL "software_popcount")
              target_compile_options(sst_${name}_${backend} PRIVATE -mno-popcnt)
            elseif(backend STREQUAL "prefetch_disabled")
              target_compile_definitions(sst_${name}_${backend} PRIVATE SST_DISABLE_PREFETCH)
            endif()
          endforeach()
        endforeach()
      else()
        message(WARNING "AVX2/POPCNT unavailable: accelerated tests skipped; generic tests remain enabled")
      endif()
    else()
      message(WARNING "Accelerated tests require native GCC/Clang on x86; generic tests remain enabled")
    endif()
  endif()
endif()
